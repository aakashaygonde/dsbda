import java.io.File;
import java.util.Scanner;

public class WeatherAnalyzer {
    public static void main(String[] args) {
        String fileName = "sample_weather.txt";
        
        double totalTemp = 0, totalDew = 0, totalWind = 0;
        int count = 0;

        try (Scanner scanner = new Scanner(new File(fileName))) {
           
            if (scanner.hasNextLine()) {
                scanner.nextLine();
            }

         
            while (scanner.hasNextLine()) {
                String line = scanner.nextLine();
              
                String[] parts = line.split(",");

                if (parts.length == 4) {
                    totalTemp += Double.parseDouble(parts[1]);
                    totalDew  += Double.parseDouble(parts[2]);
                    totalWind += Double.parseDouble(parts[3]);
                    count++;
                }
            }

    
            if (count > 0) {
                System.out.println("--- Weather Analysis Results ---");
                System.out.printf("Average Temperature: %.2f°C%n", (totalTemp / count));
                System.out.printf("Average Dew Point:   %.2f°C%n", (totalDew / count));
                System.out.printf("Average Wind Speed:  %.2f km/h%n", (totalWind / count));
            } else {
                System.out.println("No data found in the file.");
            }

        } catch (Exception e) {
            System.err.println("Error reading the file: " + e.getMessage());
        }
    }
}
