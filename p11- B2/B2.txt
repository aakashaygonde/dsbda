in File reducer.py : 

import sys

current_type = None
current_count = 0

for line in sys.stdin:

    line = line.strip()

    log_type, count = line.split('\t')

    count = int(count)

    if current_type == log_type:

        current_count += count

    else:

        if current_type:

            print(f"{current_type}\t{current_count}")

        current_type = log_type

        current_count = count

if current_type == log_type:

    print(f"{current_type}\t{current_count}")
